```bash
# Single combined invocation: walk roles/karma + roles/promlens, filter to YAML / Jinja files,
# exclude README.md, and replace the UTF-8 byte sequence for U+00A7 (0xc2 0xa7) with ASCII "sec."
LC_ALL=C find roles/karma roles/promlens -type f \
  \( -name '*.yml' -o -name '*.yaml' -o -name '*.j2' \) \
  ! -name 'README.md' \
```
