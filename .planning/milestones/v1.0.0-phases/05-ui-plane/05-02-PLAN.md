# roles/karma/defaults/main.yml
# Karma v0.130 role tunables. Override per-environment in
# inventory/<env>/group_vars/all/karma.yml.
#
# UI plane role -- D-82 exempts from telemetron_publish_default:false.
# Karma reads alerts from Alertmanager on the telemetron Docker bridge
# (http://alertmanager:9093 -- NOT localhost; D-25 audit per RESEARCH §6.2).
# Stateless: no volume, in-memory cache.
# HEALTHCHECK is explicit -- image ships none (RESEARCH §2.2).

# --- Image pin (OPS-01; UI-05) ---
# GHCR official -- NOT the lmierzwa/karma Docker Hub fork.
karma_image: ghcr.io/prymitive/karma
karma_image_tag: "v0.130"

# --- Container identity ---
karma_container_name: telemetron-karma

# --- Host port publishing (D-82 inverts D-30 for UI plane) ---
karma_publish_host: true
karma_bind_address: 0.0.0.0   # D-83

# --- Port matrix (D-84) ---
# Karma's internal listen port is 8080 (RESEARCH §2.2). Host-publish at 8082.
karma_http_port: 8082          # host-side
karma_container_port: 8080     # container-side; matches karma's listen.port default

# --- Storage layout (D-18; no volume -- Karma is stateless) ---
karma_config_dir: "{{ telemetron_config_root | default('/opt/telemetron') }}/karma"
karma_config_container_path: /etc/karma
karma_config_file_container_path: /etc/karma/karma.yaml

# --- Alertmanager connection (D-25 audit fix per RESEARCH §6.2) ---
# Telemetron uses Docker bridge DNS: http://alertmanager:9093 (NOT localhost).
karma_alertmanager_name: telemetron     # clearer than upstream `alertmanager`; RESEARCH §6.2
karma_alertmanager_host: alertmanager
karma_alertmanager_port: 9093
karma_alertmanager_proxy: true          # Karma proxies silence-add/del back to AM

# --- Alert acknowledgement (RESEARCH §2.2 -- useful homelab feature) ---
karma_ack_enabled: true
karma_ack_duration: "1h"
karma_ack_author: "karma"

# --- UI (RESEARCH §2.2 -- recommended knobs surfaced as inventory) ---
karma_ui_refresh: "30s"
karma_ui_theme: auto                    # auto / light / dark
karma_ui_minimal_group_width: 420
karma_ui_collapse_groups: collapsedOnMobile

# --- Operator extension knobs (sorted-keys per D-20) ---
karma_default_filters: []               # list of default LogQL-style filter strings
karma_labels_keep: []                   # list of label names to display in grid view
karma_extra_alertmanager_servers: []    # additional AM endpoints (for multi-AM ops; v2)

# --- Healthcheck (RESEARCH §2.2 -- image has NO built-in HEALTHCHECK; explicit required) ---
karma_healthcheck_enabled: true
karma_healthcheck_test: ["CMD-SHELL", "wget --spider -q http://localhost:{{ karma_container_port }}/health || exit 1"]
karma_healthcheck_interval: 15s
karma_healthcheck_timeout: 5s
karma_healthcheck_retries: 5
karma_healthcheck_start_period: 30s

# 60s total budget = 30 retries x 2s. Karma cold-start is fast (config parse + start HTTP).
karma_health_retries: 30
karma_health_delay: 2

# --- Restart policy (OPS-06) ---
karma_restart_policy: unless-stopped

# --- Resource limits ---
karma_memory_limit: 128m

# --- Network ---
karma_network: "{{ telemetron_network | default('telemetron') }}"

# --- Timezone (OPS-06) ---
karma_tz: "{{ telemetron_tz | default('Etc/UTC') }}"

# --- Curl image pin for verify (OPS-01) ---

```yaml

# roles/karma/tasks/main.yml
# Karma v0.130 deploy + in-network verify.
# Mirrors roles/alertmanager/tasks/main.yml shape modulo Karma specifics:
# stateless (no volume), CONFIG_FILE env var (RESEARCH §2.2 -- required so
# Karma's binary finds the config from a non-CWD path), explicit HEALTHCHECK
# (image has none -- RESEARCH §2.2), parent-dir bind mount (Gate 8).

- name: Ensure karma config directory exists (D-18 flat layout)
  ansible.builtin.file:
    path: "{{ karma_config_dir }}"
    state: directory
    mode: "0755"
  tags:
    - karma
    - karma-config

- name: Render karma.yaml from template
  ansible.builtin.template:
    src: karma.yaml.j2
    dest: "{{ karma_config_dir }}/karma.yaml"
    mode: "0644"
  notify: restart karma
  tags:
    - karma
    - karma-config

- name: Pull karma image (GHCR official, NOT Docker Hub fork)
  community.docker.docker_image:
    name: "{{ karma_image }}:{{ karma_image_tag }}"
    source: pull
    force_source: false
  tags:
    - karma

- name: Run karma container (stateless, explicit HEALTHCHECK, CONFIG_FILE env, parent-dir bind)
  community.docker.docker_container:
    name: "{{ karma_container_name }}"
    image: "{{ karma_image }}:{{ karma_image_tag }}"
    state: started
    recreate: false
    restart_policy: "{{ karma_restart_policy }}"
    memory: "{{ karma_memory_limit }}"
    networks:
      - name: "{{ karma_network }}"
        aliases:
          - "{{ karma_container_name }}"
          - karma
    # Gate 7: telemetron label stamp.
    labels:
      org.telemetron.service: telemetron
      org.telemetron.job: karma
    # D-82 + D-83 + D-84: UI plane is browser-reachable by default.
    published_ports: >-
      {{
        (
          [ karma_bind_address + ':' + karma_http_port | string + ':' + karma_container_port | string ]
          if karma_publish_host is sameas true else
          [ '127.0.0.1:' + karma_http_port | string + ':' + karma_container_port | string ]
          if karma_publish_host == '127.0.0.1' else
          []
        )
      }}
    mounts:
      # Gate 8 parent-directory bind: container sees /etc/karma/karma.yaml
      # at the path CONFIG_FILE points to.
      - source: "{{ karma_config_dir }}"
        target: "{{ karma_config_container_path }}"
        type: bind
        read_only: true
    healthcheck: "{{ karma_container_healthcheck if (karma_healthcheck_enabled | bool) else omit }}"
    env:
      # RESEARCH §2.2: Karma searches for karma.yaml in CWD by default; CWD
      # in the distroless-like image is `/`. CONFIG_FILE env var points it
      # at the parent-dir-mounted config. Without this, Karma starts with
      # empty defaults and no Alertmanager configured.
      CONFIG_FILE: "{{ karma_config_file_container_path }}"
      TZ: "{{ karma_tz }}"
  vars:
    karma_container_healthcheck:
      test: "{{ karma_healthcheck_test }}"
      interval: "{{ karma_healthcheck_interval }}"
      timeout: "{{ karma_healthcheck_timeout }}"
      retries: "{{ karma_healthcheck_retries }}"
      start_period: "{{ karma_healthcheck_start_period }}"
  tags:
    - karma
    - karma-container

- name: Flush handlers before verify (so verify sees post-restart state)
  ansible.builtin.meta: flush_handlers
  tags:
    - karma
    - karma-verify

- name: Verify karma end-to-end (HEALTHCHECK + /health + Alertmanager connection)
  ansible.builtin.include_tasks: verify.yml
  tags:
    - karma
    - karma-verify
```

# roles/karma/tasks/verify.yml
# Pre-poll Karma readiness, then exercise the public surface in-network:
# - D-10a HEALTHCHECK poll
# - /health probe (Karma's built-in endpoint)
# - root path HTML smoke (Karma UI loaded)
# - Alertmanager connection assertion (Karma's /api/v1/alerts endpoint
#   returns 200 with our alertmanager source registered)

- name: Wait for karma container HEALTHCHECK to report healthy
  community.docker.docker_container_info:
    name: "{{ karma_container_name }}"
  register: karma_health_check
  until: >-
    karma_health_check.container is defined
    and karma_health_check.container.State is defined
    and karma_health_check.container.State.Health is defined
    and karma_health_check.container.State.Health.Status == 'healthy'
  retries: "{{ karma_health_retries }}"
  delay: "{{ karma_health_delay }}"
  changed_when: false
  when: karma_healthcheck_enabled | bool
  tags:
    - karma
    - karma-verify

- name: Curl-probe /health in-network (Karma source-confirmed endpoint per RESEARCH §2.2)
  community.docker.docker_container:
    name: "{{ karma_container_name }}-verify-health"
    image: "{{ karma_curl_image }}:{{ karma_curl_image_tag }}"
    state: started
    detach: false
    auto_remove: true
    recreate: true
    cleanup: true
    networks:
      - name: "{{ karma_network }}"
    entrypoint: ["/bin/sh", "-c"]
    command:
      - >-
        curl -fsS http://{{ karma_container_name }}:{{ karma_container_port }}/health
        && echo "karma_health_ok"
  changed_when: false
  tags:
    - karma
    - karma-verify

- name: Curl-probe root path (Karma UI HTML smoke)
  community.docker.docker_container:
    name: "{{ karma_container_name }}-verify-ui"
    image: "{{ karma_curl_image }}:{{ karma_curl_image_tag }}"
    state: started
    detach: false
    auto_remove: true
    recreate: true
    cleanup: true
    networks:
      - name: "{{ karma_network }}"
    entrypoint: ["/bin/sh", "-c"]
    command:
      - >-
        curl -fsS http://{{ karma_container_name }}:{{ karma_container_port }}/
        | grep -qiE 'karma|<html' || { echo "karma UI HTML smoke FAIL" ; exit 1 ; } ;
        echo "karma_ui_ok"
  changed_when: false
  tags:
    - karma
    - karma-verify

# UI-05 wiring assertion: Karma reports our alertmanager as a configured upstream.
# Karma's /alerts.json (or /api/v1/alerts) endpoint exposes the alertmanager
# source name in the response payload. Assert the configured name appears.
# Poll with until: -- Karma's first Alertmanager poll is async after startup.
- name: Karma -> Alertmanager connection assertion (poll until source registered)
  community.docker.docker_container:
